# Attention Mechanism Tests
cmake_minimum_required(VERSION 3.20)

function(dpan_learning_test name)
    add_executable(${name} ${name}.cpp)
    target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/tests/learning)
    target_link_libraries(${name} dpan_learning GTest::gtest_main)
    gtest_discover_tests(${name})
endfunction()
